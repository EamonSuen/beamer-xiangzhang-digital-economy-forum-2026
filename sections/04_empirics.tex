\section{实证证据}

\begin{frame}{样本与识别策略}
  我们的样本是 2009--2023 年的 A 股上市公司。

  \vspace{0.4em}
  识别策略是交叠双重差分，核心设定是：
  \[
    Y_{it}=\alpha_i+\gamma_t+\beta D_{it}
    +\Gamma X_i \times f(t)+\varepsilon_{it}.
  \]

  \begin{itemize}
    \item $D_{it}$ 是企业 $i$ 在 $t$ 年是否已通过贯标认证；
    \item $\alpha_i$ 和 $\gamma_t$ 分别控制企业固定效应和年份固定效应；
    \item 识别来自于\emph{同一家企业贯标前后的投资结构变化}，同时由尚未贯标的企业提供对照。
  \end{itemize}

  \vspace{0.4em}
  $X_i\times f(t)$ 是事前特征与年份三次多项式的交乘——这么做的目的是在控制事前差异的同时，避免把政策后的变量当成坏控制。
\end{frame}

\begin{frame}{为什么不能假设"无预期"？}
  前面提到过，两化融合不是突发冲击。如果忽略预期效应，事前系数就会被污染。

  \vspace{0.4em}
  我们的参考既有文献的处理方式（Nagengast and Yotov，2025；Bertolotti，2026）：
  \begin{itemize}
    \item 把有效处理起点前移到认证通过日之前若干期；
    \item 用事前趋势诊断和有限预期窗口，把预期效应和实际处理效应分开。
  \end{itemize}

  \vspace{0.4em}
  具体来说，我们估计：
  \[
    Y_{it}=\alpha_i+\gamma_t
    +\sum_{l=-K_1}^{-2}\mu_lD_{it}^{l}
    +\sum_{l=0}^{K_2}\mu_lD_{it}^{l}
    +\Gamma X_i\times f(t)+\varepsilon_{it}.
  \]
\end{frame}

\begin{frame}{机制一：怎么度量"抵押囤积"？}
  \small
  模型预测：贯标对 CapEX 的抑制\emph{不应均匀分布}，而应集中在事前囤积超额资本的企业。要把"为生产持有的资本"与"为抵押持有的资本"区分开——前者由生产基本面决定，后者则把资本推到基本面之上。

  \begin{methodbox}{事前囤积楔子 PH（贯标前窗口估计）}
    \footnotesize
    \vspace{-0.4em}
    \[
      \ln K_{it}=\alpha_0+\alpha_1\ln Y_{it}+\alpha_2\ln L_{it}
      +\alpha_3\,\text{SalesGrowth}_{it}+\gamma_j+\gamma_t+u_{it}
    \]
    \vspace{-1.1em}
    \begin{itemize}\setlength\itemsep{0pt}
      \item $K_{it}=$ 固定资产 + 在建工程 + 投资性房地产；残差 $\hat u_{it}$ = 多余资本；
      \item $\text{PH}_i=$ 企业事前 $\hat u_{it}$ 均值（标准化），数值越大囤积越严重。
    \end{itemize}
    \vspace{-0.2em}
    {\scriptsize 设计：参考Richardson (2006) 过度投资残差 以及 Chaney et al.\ (2012) 事前暴露度。}
  \end{methodbox}

  \begin{robustbox}{排除"技术落后"替代解释}
    \footnotesize
    控制企业事前 TFP 后，TFP 系数显著为负、$R^2$ 从 0.74 升至 0.77——技术越高效的企业所需资本越少。\textbf{PH 捕捉的不是落后产能。}
  \end{robustbox}
\end{frame}

\begin{frame}{机制一：囤积越多的企业，收缩越剧烈}
  \input{tables/pledge-hoarding}

  \begin{findingbox}{机制成立要满足三个层层递进的可证伪推论}
    \scriptsize
    \begin{itemize}\setlength\itemsep{1pt}
      \item \textbf{(1) PH 有经济含义}\,---\,列 (1)：PH 与抵押率显著正相关，超额资本企业更依赖抵押融资；
      \item \textbf{(2) 收缩集中在高 PH}\,---\,列 (2)：PH 每高一个标准差，CapEX 额外收缩 \textbf{1.7\,pp}，约基准的\emph{三倍}；
      \item \textbf{(3) 融资结构同步去抵押化}\,---\,列 (3)：高 PH 企业贯标后抵押率显著下降，方向与实物出清一致。
    \end{itemize}
    \vspace{-0.2em}
    \centering $\Rightarrow$ 实物资本收缩 $=$ \emph{抵押动机弱化驱动的主动出清}，而非被动紧缩。
  \end{findingbox}
\end{frame}

\begin{frame}{出清发生在哪里？边际投资，不是存量甩卖}
  \begin{itemize}
    \item 把有形资本按抵押属性拆开后，交互项的负效应主要集中在\emph{在建工程}——抵押属性最弱的科目；
    \item 固定资产、投资性房地产等高粘性存量资产，反应并不显著。
  \end{itemize}

  \vspace{0.5em}
  \begin{findingbox}{这个细节很重要}
    囤积型企业\emph{停止追加新的资本形成}，出清通过控制边际投资流量实现，而不是通过甩卖既有存量——这和 Shleifer--Vishny（1992）讲的资产流动性折价逻辑是一致的。
  \end{findingbox}
\end{frame}

\begin{frame}{机制二：信息替代抵押——企业端证据}
  现在看模型的第三个预测：信息替代抵押的逻辑在数据上是否成立。

  \vspace{0.4em}
  我们构造了一个\emph{剩余抵押空间}指标：
  \[
    \text{Slack}_{it}
    =
    \overline{\text{MR}}_{j,p90}^{2011}
    -
    \text{MR}_{it}.
  \]

  \begin{itemize}
    \item $\text{MR}_{it}$ 是企业实际的抵押率，$\overline{\text{MR}}_{j,p90}^{2011}$ 是行业在 2011 年的第 90 百分位；
    \item 差值越大，说明企业离行业抵押天花板越远，传统抵押空间越充裕；
    \item 差值越小甚至为负——说明企业已经快要触及传统抵押上限了。
  \end{itemize}
\end{frame}

\begin{frame}{机制二：银行端理论框架——竞争结构如何决定银行的信息策略}
  \textbf{Petersen \& Rajan (1995); Berger \& Udell (2002)} 经典框架：信贷市场的竞争结构决定银行的信息生产策略——

  \vspace{0.4em}
  \begin{itemize}
    \setlength\itemsep{0.5em}
    \item \emph{高集中度市场}：银行凭借市场势力\textbf{内部化}未来租金，有激励通过长期银企关系生产\textbf{软信息}；
    \item \emph{高竞争市场}：跨期租金分享机制瓦解，银行只能依赖可观测、可验证的\textbf{硬信息}与\textbf{实物抵押}进行放贷。
  \end{itemize}

  \vspace{0.4em}
  \begin{methodbox}{把贯标视为"硬信息供给冲击"——可证伪推论}
    若信息硬化是核心渠道，贯标的信贷扩张效应应集中在\emph{软信息相对稀缺的高竞争市场}；若仅通过信号显示 / 政府背书 / 补贴等替代渠道发挥作用，效应不应与银行竞争结构呈现这种系统替代关系。
  \end{methodbox}
\end{frame}

\begin{frame}{效率含义：为什么 CapEX 下降可能是效率改善？}
  最后看效率。关键不是实物投资规模本身，而是资本是否从\emph{抵押驱动的低效存量}转向\emph{边际回报更高的项目}。

  \begin{methodbox}{理论推演}
    \footnotesize
    在存在融资摩擦时，资本边际产出偏离效率基准：受约束的高回报企业资本不足，MRPK 偏高；抵押能力强、囤积有形资产的企业资本过多，MRPK 偏低。数字化信息替代抵押后，维持冗余抵押品的融资动机下降，资本配置应向效率基准回归。
  \end{methodbox}

  \vspace{-0.3em}
  \[
    MRPK_{it}
    =
    \frac{\partial \text{Revenue}_{it}}{\partial K_{it}}
    =
    \alpha_K^j\cdot\frac{\text{Revenue}_{it}}{K_{it}}.
  \]
  \vspace{-0.8em}

  \begin{enumerate}\setlength\itemsep{0.25em}
    \item \textbf{存量错配下降：}高 MRPK 企业获得融资修复后，边际回报应向行业均值回落；
    \item \textbf{增量筛选改善：}即使总 CapEX 下降，投资也应更贴近 Tobin's $Q$ 所代表的成长机会；
    \item \textbf{冗余产能出清：}资本越老，越容易落入抵押囤积区域，贯标后的收缩应更强。
  \end{enumerate}
\end{frame}
